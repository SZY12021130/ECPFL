\section{Consensus Selection Strategies}
\label{sec:consensus}

Let $\bar{\mathbf{v}}^{(t)}$ denote the global vote tally vector, where each entry
\begin{equation}
\bar{v}_j^{(t)} = \sum_{i \in \mathcal{S}^{(t)}} v_{i,j}^{(t)}
\end{equation}
counts how many active clients voted for pack $j$ at round $t$. The server obtains the consensus set $\mathcal{K}^{(t)}$ using one of the following rules.

\begin{itemize}
    \item \textbf{Union (\textsc{ECPFL-UN}):} retains every pack that receives at least one vote,
    \begin{equation}
        \mathcal{K}^{(t)} = \Bigl\{ j \in [J] \;\Big|\; \bar{v}_j^{(t)} \ge 1 \Bigr\}.
    \end{equation}

    \item \textbf{Rank (\textsc{ECPFL-RK-}$\theta$):} sorts packs by their global vote tallies in descending order and retains the top-$\lfloor \theta J \rfloor$ packs, where $\theta \in (0,1]$,
    \begin{equation}
        \mathcal{K}^{(t)} = \Bigl\{ \pi_\ell \in [J] \;\Big|\; 1 \le \ell \le \lfloor \theta J \rfloor,\; \bar{v}_{\pi_1}^{(t)} \ge \bar{v}_{\pi_2}^{(t)} \ge \dots \ge \bar{v}_{\pi_J}^{(t)} \Bigr\}.
    \end{equation}

    \item \textbf{Threshold (\textsc{ECPFL-TH-}$\theta$):} retains a pack only if its accumulated votes reach a proportion $\theta \in (0,1]$ of the active clients,
    \begin{equation}
        \mathcal{K}^{(t)} = \Bigl\{ j \in [J] \;\Big|\; \bar{v}_j^{(t)} \ge \theta \cdot |\mathcal{S}^{(t)}| \Bigr\}.
    \end{equation}
\end{itemize}
