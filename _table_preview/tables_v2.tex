\midrule
FedAvg+CKKS & 1 & $\Theta(d\log Q)$ & any subset & shared $sk$ \\

ESAFL       & 1 & $\Theta(d\log Q)$ & \textbf{all $N$ clients} & trusted AS (one-time) \\
MASER       & 2 (vote $\to$ ct) $+$ shares & $\Theta(d)+\Theta(\zeta J n\log Q)$ (ct \& share) & \textbf{all $N$ clients} & trusted key manager \\
SAMK        & 1 & $\Theta\big(d\log Q+\tfrac{Nd}{n}\log\mathcal{N}\big)$ & \textbf{any non-empty subset} & \textbf{none} \\
\midrule
\rowcolor{blue!5}
ECPFL (ours) & 1 & $\Theta\big((k{+}1)n\log Q\big)$ & any $\widehat{\mathcal{S}}^{(t)}$ with $|\widehat{\mathcal{S}}^{(t)}|\ge\tau$ & offline KGC (one-time shares) \\
\bottomrule
\end{tabular}%
}
\par\vspace{2pt}
{\footnotesize $J=\lceil d/n\rceil$; $\zeta$: MASER keep-ratio; $k_s$: FedPHE sketch dimension; $N$: clients (SAMK interpolation points); $\mathcal{N}$: Paillier modulus. MASER uploads plaintext masks in an extra voting round; its partial-decryption shares are ciphertext-sized, hence counted twice.}
